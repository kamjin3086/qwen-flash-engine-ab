globalThis.fetch=async function(input,init){

 if(url?.startsWith(base)&&url.endsWith('/chat/completions')){
  if(requestN>=12||outputTokens>=65536)throw Error('generation budget exhausted');

  if(!cpu.passed)throw Error('CPU gate failed before model request');
  requestN++; const prefix=path.join(out,`request-${String(requestN).padStart(2,'0')}`);
  const body=JSON.parse(init.body);
  const declared=(body.tools||[]).map(t=>t.function?.name).sort();
  if(JSON.stringify(declared)!==JSON.stringify(['edit_file','inspect_page','read_file','write_file']))throw Error('harness invalid: expected all four confined tools before sending');

  fs.writeFileSync(prefix+'.json',JSON.stringify(body,null,2));
