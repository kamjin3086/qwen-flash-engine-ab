function makeInstanced(pos,col,glow,geo){

  for(let i=0;i<n;i++){
    p.set(pos[i*3],pos[i*3+1],pos[i*3+2]);
    m.compose(p,q,sc);mesh.setMatrixAt(i,m);

  mesh.geometry.setAttribute('aColor',new THREE.InstancedBufferAttribute(colArr,3));
  mesh.geometry.setAttribute('aGlow',new THREE.InstancedBufferAttribute(glowArr,3));

function buildGround(){
  for(let x=-R;x<=R;x++)for(let z=-R;z<=R;z++){
    const d=Math.hypot(x,z);
    if(d>R+2)continue;
    if(d>R){ if(rng()>.5)continue; }
    let col;
    if(isPond(x,z)){
      const depth=1-smooth(11.5,13.6,Math.sqrt(dist2(x,z,-18,16)));
      col=vary(lerpC(C.water[0],C.waterDeep[0],depth),.06);
    }else if(isPath(x,z)){col=vary(pick(C.path),.06);}
    else if(isPlaza(x,z)){col=vary(pick(C.path),.05);}
    else if(isTree(x,z)){col=vary(pick(C.grassDark),.06);}
    else if(isRock(x,z)){col=vary(pick(C.grassDark),.06);}
    else if(isFence(x,z)){col=vary(pick(C.grassDark),.06);}
    else if(isBamboo(x,z)){col=vary(pick(C.grassDark),.06);}
    else{col=vary(pick(C.grass),.07);}
    builder.add(x,0,z,col);
  }
}

/* ---- water ---- */
function buildWater(){
  for(let x=-R;x<=R;x++)for(let z=-R;z<=R;z++){
    if(!isPond(x,z))continue;
    const d=Math.sqrt(dist2(x,z,-18,16));
    const depth=1-smooth(11.5,13.6,d);
    const col=vary(lerpC(C.water[0],C.waterDeep[0],depth),.05);
    builder.add(x,-.35,z,col);
  }
}

  for(let x=-10;x<=10;x++)for(let z=-10;z<=10;z++){
    const d=Math.max(Math.abs(x),Math.abs(z));
    if(d<=10)P(x,1,z,vary(pick(C.stone),.05));
    if(d<=8)P(x,2,z,vary(pick(C.stone),.05));
  }
  // steps
  for(let s=0;s<3;s++)for(let x=-3;x<=3;x++)P(x,1+s,11-s,vary(pick(C.stone),.05));
  // central pillar
  for(let y=3;y<=40;y++)P(0,y,0,vary(pick(C.pagodaWood),.05));
  // floors
  const floors=[{y:3,r:8},{y:9,r:7},{y:15,r:6},{y:21,r:5},{y:27,r:4}];
  for(const f of floors){
    const r=f.r;
    for(let x=-r;x<=r;x++)for(let z=-r;z<=r;z++){
      const d=Math.max(Math.abs(x),Math.abs(z));
      if(d>r)continue;
      if(d===r){P(x,f.y,z,vary(pick(C.pagodaRed),.05));P(x,f.y+1,z,vary(pick(C.pagodaRed),.05));}
      else if(d===r-1){
        const corner=(Math.abs(x)===r-1&&Math.abs(z)===r-1);
        if(corner)P(x,f.y,z,vary(pick(C.pagodaWood),.05));
        else if((x+z)%2===0)P(x,f.y,z,vary(pick(C.pagodaDark),.04));
        else P(x,f.y,z,vary(pick(C.pagodaGold),.05),.15);
      }else if(d===r-2){P(x,f.y,z,vary(pick(C.pagodaWood),.05));}
    }
    // corner posts
    for(const sx of[-1,1])for(const sz of[-1,1])P(sx*r,f.y+2,sz*r,vary(pick(C.pagodaWood),.05));
    // roof
    const ry=f.y+3;
    for(let x=-r-2;x<=r+2;x++)for(let z=-r-2;z<=r+2;z++){
      const d=Math.max(Math.abs(x),Math.abs(z));
      if(d>r+2)continue;
      if(d===r+2){
        const corner=(Math.abs(x)===r+2&&Math.abs(z)===r+2);
        P(x,ry+(corner?1:0),z,vary(pick(C.pagodaRoof),.05));
      }else if(d===r+1)P(x,ry,z,vary(pick(C.pagodaRoof),.05));
      else if(d===r)P(x,ry,z,vary(pick(C.pagodaRoofGold),.05),.1);
      else if(d===r-1)P(x,ry,z,vary(pick(C.pagodaRoof),.05));
    }
  }
  // spire
  for(let y=30;y<=38;y++)P(0,y,0,vary(pick(C.pagodaGold),.05),.2);
  for(let y=31;y<=37;y+=2)for(let x=-1;x<=1;x++)for(let z=-1;z<=1;z++)
    if(Math.max(Math.abs(x),Math.abs(z))===1)P(x,y,z,vary(pick(C.pagodaGold),.05),.15);
  P(0,39,0,vary(pick(C.pagodaGold),.05),.4);
  P(0,40,0,vary(pick(C.pagodaGold),.05),.5);
}

/* ---- trees ---- */
function buildTree(t){
  const P=(x,y,z,c,g=0)=>builder.add(x,y,z,c,g);
  const th=t.type==='pine'?ri(6,9):ri(5,8);
  for(let y=1;y<=th;y++)P(t.x,y,t.z,vary(pick(C.trunk),.06));
  if(t.type==='pine'){
    let cy=th+1;
    for(let layer=0;layer<4;layer++){
      const rad=Math.max(1,4-layer);
      for(let dx=-rad;dx<=rad;dx++)for(let dz=-rad;dz<=rad;dz++){
        if(Math.abs(dx)+Math.abs(dz)>rad+1)continue;
        if(rng()<.12)continue;
        P(t.x+dx,cy,dz+0,vary(pick(C.pine),.08));
      }
      cy+=2;
    }
    P(t.x,cy,t.z,vary(pick(C.pine),.08));
  }else{
    const rad=t.type==='sakura'?4:3.6;
    const pal=t.type==='sakura'?C.sakura:(t.type==='maple'?C.maple:C.leaf);
    const deep=t.type==='sakura'?C.sakuraDeep:(t.type==='maple'?C.maple2:C.leafDeep);
    for(let dx=-rad;dx<=rad;dx++)for(let dy=-rad;dy<=rad;dy++)for(let dz=-rad;dz<=rad;dz++){
      const d=Math.sqrt(dx*dx+dy*dy+dz*dz);
      if(d>rad)continue;
      if(d>rad-1.2&&rng()<.4)continue;
      const y=th+2+dy;
      if(y<1)continue;
      const edge=d>rad-1.4;
      P(t.x+dx,y,t.z+dz,vary(edge?pick(pal):pick(rng()<.3?deep:pal),.07));
    }
  }
}

function updateMain(dt,t){

function updatePetals(dt,t){
  petalAmt+=(petalTarget-petalAmt)*Math.min(1,dt*3);
  const mesh=petalGroup.children[0],ud=mesh.userData;
  const aC=mesh.geometry.attributes.aColor.array,base=ud.base;
  const wind=Math.sin(t*.3)*.6+.4;
  for(let i=0;i<petalData.length;i++){
    const p=petalData[i];
    p.y-=p.vy*petalAmt*60*dt;
    if(p.y<.4){p.y=rnd(30,44);p.x=rnd(-46,46);p.z=rnd(-46,46);}
    const px=p.x+Math.sin(t*p.sp+p.ph)*p.amp+wind*2;
    const pz=p.z+Math.cos(t*p.sp*.8+p.ph)*p.amp;
    _e.set(t*p.sp*2+p.ph,t*p.sp*1.3,p.ph);_q.setFromEuler(_e);
    _v.set(p.x,p.y,p.z);
    _m.compose(_v,_q,new THREE.Vector3(1,1,1));
    mesh.setMatrixAt(i,_m);
    const a=clamp(petalAmt*1.1,0,1);
    aC[i*3]=base[i*3]*a;aC[i*3+1]=base[i*3+1]*a;aC[i*3+2]=base[i*3+2]*a;
  }
  mesh.instanceMatrix.needsUpdate=true;
  mesh.geometry.attributes.aColor.needsUpdate=true;

A few things worth knowing:

- **The scene** is a five-tier pagoda on a stone platform, ringed by cherry-blossom, maple and pine trees, with a koi pond (swimming koi + lily pads), a red torii gate, stone lanterns that glow at night, an arched bridge, a bamboo grove, iris/peony beds, a rock garden, and a fence — all under a red moon and drifting clouds.
- **Time of day** (Day / Dusk / Night) smoothly interpolates the sky gradient, sun color/position, fog, exposure, star/moon opacity, and makes the lanterns and pagoda gold trim glow.
- **Petals** fall and drift with wind; **Rebuild** replays the grow-in animation; **Auto/Manual** toggles the orbiting camera.
- Everything is instanced voxels, so it stays smooth even with ~40k cubes. If you want a denser/thinner garden, tweak the `R` value, the `trees`/`rocks` arrays, or the petal `N`.
