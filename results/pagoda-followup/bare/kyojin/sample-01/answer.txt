Here's a complete, self-contained voxel garden — a five-tiered pagoda, a koi pond with a stone bridge, cherry blossoms that drift on the wind, a torii gate, a lantern, a red bridge, and a day/night cycle with fireflies. Everything is generated procedurally from a tiny voxel engine (no external assets), so you can paste the whole thing into one `.html` file and open it.

/* ============================================================
   Tiny voxel engine — canvas 2D, procedural, no dependencies
   ============================================================ */
const canvas = document.getElementById('scene');

/* ---- color helpers ---- */
const clamp=(v,a,b)=>v<a?a:v>b?b:v;
const lerp=(a,b,t)=>a+(b-a)*t;
const lerp3=(a,b,t)=>[lerp(a[0],b[0],t),lerp(a[1],b[1],t),lerp(a[2],b[2],t)];
const mix=(a,b,t)=>lerp3(a,b,t);
const shade=(c,f)=>[clamp(c[0]*f,0,255),clamp(c[1]*f,0,255),clamp(c[2]*f,0,255)];
const rgb=c=>`rgb(${c[0]|0},${c[1]|0},${c[2]|0})`;
const rgba=(c,a)=>`rgba(${c[0]|0},${c[1]|0},${c[2]|0},${a})`;
const rand=(a,b)=>a+Math.random()*(b-a);
const hash=(x,y,z)=>{let h=(x*374761393+y*668265263+z*1274126177)>>>0;h=(h^(h>>13))*1274126177>>>0;return (h^(h>>16))>>>0;};
const jitter=(c,amt)=>{const d=((hash(c[0],c[1],c[2])%1000)/1000-0.5)*amt;return shade(c,1+d);};

/* ---- palettes ---- */

/* ---- voxel grid ---- */
const grid=new Map();
const key=(x,y,z)=>x+','+y+','+z;
const has=(x,y,z)=>grid.has(key(x,y,z));
function setVoxel(x,y,z,c){x=Math.round(x);y=Math.round(y);z=Math.round(z);grid.set(key(x,y,z),c);}
function fillBox(x0,x1,y0,y1,z0,z1,c){for(let x=x0;x<=x1;x++)for(let y=y0;y<=y1;y++)for(let z=z0;z<=z1;z++)setVoxel(x,y,z,c);}
function fillRect(x0,x1,z0,z1,y,c){for(let x=x0;x<=x1;x++)for(let z=z0;z<=z1;z++)setVoxel(x,y,z,c);}
function fillDisc(cx,cz,r,y,c){for(let x=Math.floor(cx-r);x<=Math.ceil(cx+r);x++)for(let z=Math.floor(cz-r);z<=Math.ceil(cz+r);z++){const dx=x-cx,dz=z-cz;if(dx*dx+dz*dz<=r*r)setVoxel(x,y,z,c);}}
function fillRing(cx,cz,r,y,c){for(let x=Math.floor(cx-r);x<=Math.ceil(cx+r);x++)for(let z=Math.floor(cz-r);z<=Math.ceil(cz+r);z++){const dx=x-cx,dz=z-cz,d=Math.sqrt(dx*dx+dz*dz);if(d<=r&&d>r-1.15)setVoxel(x,y,z,c);}}
function fillCyl(cx,cz,r,y0,y1,c){for(let y=y0;y<=y1;y++)fillDisc(cx,cz,r,y,c);}
function fillCylRing(cx,cz,r,y0,y1,c){for(let y=y0;y<=y1;y++)fillRing(cx,cz,r,y,c);}
function fillCone(cx,cz,r,y0,h,c){for(let i=0;i<h;i++){const y=y0+i,rr=r*(1-i/h);fillDisc(cx,cz,Math.max(0,rr),y,c);}}
function fillPyramid(cx,cz,r,y0,h,c){for(let i=0;i<h;i++){const y=y0+i,rr=r*(1-i/h);fillRect(Math.round(cx-rr),Math.round(cx+rr),Math.round(cz-rr),Math.round(cz+rr),y,c);}}
function fillSphere(cx,cy,cz,r,c){for(let x=Math.floor(cx-r);x<=Math.ceil(cx+r);x++)for(let y=Math.floor(cy-r);y<=Math.ceil(cy+r);y++)for(let z=Math.floor(cz-r);z<=Math.ceil(cz+r);z++){const dx=x-cx,dy=y-cy,dz=z-cz;if(dx*dx+dy*dy+dz*dz<=r*r)setVoxel(x,y,z,c);}}
function fillLine(x0,y0,z0,x1,y1,z1,c){const n=Math.max(Math.abs(x1-x0),Math.abs(y1-y0),Math.abs(z1-z0));for(let i=0;i<=n;i++)setVoxel(Math.round(lerp(x0,x1,i/n)),Math.round(lerp(y0,y1,i/n)),Math.round(lerp(z0,z1,i/n)),c);}

function buildGround(){
  for(let x=GROUND.x0;x<=GROUND.x1;x++)for(let z=GROUND.z0;z<=GROUND.z1;z++){
    if(inPond(x,z))continue;
    const h=hash(x,0,z);
    let c=h%11===0?P.grassD:h%9===0?P.grassL:P.grass;
    if(h%53===0)c=P.leafA; else if(h%61===0)c=P.leafB;
    setVoxel(x,GROUND_Y,z,c);
  }
}
function buildPond(){
  for(let x=POND.cx-9;x<=POND.cx+9;x++)for(let z=POND.cz-9;z<=POND.cz+9;z++){
    if(!inPond(x,z))continue;
    setVoxel(x,GROUND_Y,z,P.dirtD);
    setVoxel(x,POND_TOP,z,P.water);
  }
  fillRing(POND.cx,POND.cz,POND.r+0.6,POND_TOP,P.stone);
  fillRing(POND.cx,POND.cz,POND.r+0.6,POND_TOP+1,P.stoneL);
}

function project(x,y,z){
  const dx=x-cam.tx,dy=y-cam.ty,dz=z-cam.tz;
  const rx=dx*cosY-dz*sinY, rz=dx*sinY+dz*cosY;
  const ry=dy*cosP-rz*sinP, rz2=dy*sinP+rz*cosP;
  const depth=rz2+cam.dist;
  return {sx:cx0+rx*scale, sy:cy0-ry*scale, depth};
}

A few things worth knowing:

- **Everything is procedural** — the pagoda, torii, koi pond, stone/red bridges, lanterns, cherry trees, pines, bamboo, rocks, and flower beds are all built from a small voxel toolkit (`fillBox`, `fillCyl`, `fillCone`, `fillSphere`, etc.), then rendered with a painter's-algorithm depth sort and per-face directional lighting.
- **It's alive**: petals drift and tumble on the wind, the pond water shimmers, lanterns glow, and the **day/night** buttons (or keys `1–4`) smoothly re-light the whole scene — night brings a moon, stars, and fireflies.
- **Controls**: drag to orbit, scroll to zoom, `Space` to pause the auto-orbit, `R` to reset, and the sliders/toggles in the panel.

If you want it denser, bump the petal count in `initParticles()`, add more `buildBlossomTree`/`buildTree` calls, or widen the `GROUND` bounds.
