Here is a complete, single-file HTML solution using **Three.js** (loaded via CDN) to render a detailed voxel scene. It features a procedurally generated 5-story pagoda, cherry blossom trees, a koi pond, stone lanterns, and dynamic falling petals.

You can save the code below as an `.html` file and open it in any modern browser.

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Voxel Pagoda Garden</title>
    <style>
        body { margin: 0; overflow: hidden; background-color: #87CEEB; font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; }
        #canvas-container { width: 100vw; height: 100vh; display: block; }
        #ui {
            position: absolute; top: 20px; left: 20px; color: white;
            text-shadow: 0 2px 4px rgba(0,0,0,0.5);
            pointer-events: none;
        }
        h1 { margin: 0; font-size: 24px; font-weight: 300; letter-spacing: 2px; }
        p { margin: 5px 0 0 0; font-size: 14px; opacity: 0.8; }
        #controls {
            position: absolute; bottom: 20px; left: 20px; color: white;
            background: rgba(0,0,0,0.5); padding: 10px; border-radius: 8px;
            font-size: 12px; pointer-events: none;
        }
    </style>
    <!-- Load Three.js and OrbitControls from CDN -->
    <script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/controls/OrbitControls.js"></script>
</head>
<body>

<div id="ui">
    <h1>Voxel Pagoda Garden</h1>
    <p>Drag to rotate | Scroll to zoom</p>
</div>
<div id="controls">
    Scene: Procedural Voxel Generation | Renderer: WebGL
</div>
<div id="canvas-container"></div>

<script>
    // --- CONFIGURATION & PALETTE ---
    const COLORS = {
        woodDark: 0x5d4037,
        woodLight: 0x8d6e63,
        roofRed: 0xc62828,
        roofDark: 0x8e0000,
        gold: 0xffd700,
        white: 0xffffff,
        trunk: 0x5d4037,
        leafGreen: 0x4caf50,
        leafDark: 0x2e7d32,
        cherryPink: 0xf48fb1,
        cherryLight: 0xffc1e3,
        stoneGray: 0x9e9e9e,
        stoneDark: 0x616161,
        water: 0x4fc3f7,
        waterDeep: 0x0288d1,
        grass: 0x66bb6a,
        grassDark: 0x388e3c,
        path: 0xbcaaa4,
        lanternStone: 0x757575,
        lanternLight: 0xffeb3b
    };

    // --- SCENE SETUP ---
    const container = document.getElementById('canvas-container');
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x87CEEB); // Sky blue
    scene.fog = new THREE.Fog(0x87CEEB, 20, 60);

    const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
    camera.position.set(25, 20, 30);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    container.appendChild(renderer.domElement);

    const controls = new THREE.OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.05;
    controls.maxPolarAngle = Math.PI / 2 - 0.1; // Don't go below ground
    controls.minDistance = 10;
    controls.maxDistance = 60;

    // --- LIGHTING ---
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
    scene.add(ambientLight);

    const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
    dirLight.position.set(10, 20, 10);
    dirLight.castShadow = true;
    dirLight.shadow.mapSize.width = 2048;
    dirLight.shadow.mapSize.height = 2048;
    dirLight.shadow.camera.near = 0.5;
    dirLight.shadow.camera.far = 50;
    const d = 20;
    dirLight.shadow.camera.left = -d;
    dirLight.shadow.camera.right = d;
    dirLight.shadow.camera.top = d;
    dirLight.shadow.camera.bottom = -d;
    scene.add(dirLight);

    // --- VOXEL HELPERS ---
    // We use InstancedMesh for performance, but for simplicity in this demo, 
    // we will group meshes. For a huge scene, InstancedMesh is better.
    // Here we use a simple BoxGeometry shared across many meshes.
    
    const boxGeo = new THREE.BoxGeometry(1, 1, 1);
    
    // Cache materials to reduce draw calls slightly
    const materials = {};
    function getMaterial(color) {
        if (!materials[color]) {
            materials[color] = new THREE.MeshLambertMaterial({ color: color });
        }
        return materials[color];
    }

    function createVoxel(x, y, z, color, castShadow = true, receiveShadow = true) {
        const mesh = new THREE.Mesh(boxGeo, getMaterial(color));
        mesh.position.set(x, y, z);
        mesh.castShadow = castShadow;
        mesh.receiveShadow = receiveShadow;
        return mesh;
    }

    // --- BUILDING FUNCTIONS ---

    function buildPagoda(scene) {
        const pagodaGroup = new THREE.Group();
        
        // Base Platform
        for (let x = -3; x <= 3; x++) {
            for (let z = -3; z <= 3; z++) {
                pagodaGroup.add(createVoxel(x, 0, z, COLORS.stoneDark));
            }
        }

        // Main Structure
        const levels = 5;
        let currentY = 1;
        let currentWidth = 5; // Width of the main body at this level

        for (let i = 0; i < levels; i++) {
            // Body of the level
            const halfW = Math.floor(currentWidth / 2);
            for (let x = -halfW; x <= halfW; x++) {
                for (let z = -halfW; z <= halfW; z++) {
                    // Corner pillars are darker
                    const isCorner = (Math.abs(x) === halfW && Math.abs(z) === halfW);
                    const color = isCorner ? COLORS.woodDark : COLORS.woodLight;
                    
                    // Add some variation/texture
                    if (Math.random() > 0.9) {
                        pagodaGroup.add(createVoxel(x, currentY, z, COLORS.woodDark));
                    } else {
                        pagodaGroup.add(createVoxel(x, currentY, z, color));
                    }
                }
            }

            // Roof
            // Roof extends 1 block beyond the body on each side
            const roofW = currentWidth + 2;
            const halfRW = Math.floor(roofW / 2);
            
            // Roof Layer 1 (Main slope)
            for (let x = -halfRW; x <= halfRW; x++) {
                for (let z = -halfRW; z <= halfRW; z++) {
                    // Only draw the perimeter for the roof to save geometry, 
                    // or fill it if we want solid blocks. Let's fill it for solidity.
                    pagodaGroup.add(createVoxel(x, currentY + 1, z, COLORS.roofRed));
                }
            }
            
            // Roof Layer 2 (Smaller, darker edge)
            const roofW2 = roofW - 1;
            const halfRW2 = Math.floor(roofW2 / 2);
            for (let x = -halfRW2; x <= halfRW2; x++) {
                for (let z = -halfRW2; z <= halfRW2; z++) {
                    pagodaGroup.add(createVoxel(x, currentY + 2, z, COLORS.roofDark));
                }
            }

            // Decorative Gold Trim on corners of roof
            const corners = [
                [-halfRW, -halfRW], [halfRW, -halfRW], 
                [-halfRW, halfRW], [halfRW, halfRW]
            ];
            corners.forEach(([cx, cz]) => {
                pagodaGroup.add(createVoxel(cx, currentY + 1, cz, COLORS.gold));
            });

            currentY += 3; // Move up for next level
            currentWidth -= 1; // Narrower as we go up
        }

        // Spire
        for (let y = 0; y < 4; y++) {
            pagodaGroup.add(createVoxel(0, currentY + y, 0, COLORS.gold));
        }
        pagodaGroup.add(createVoxel(0, currentY + 4, 0, COLORS.gold)); // Tip

        scene.add(pagodaGroup);
    }

    function buildTree(scene, x, z, type) {
        const treeGroup = new THREE.Group();
        treeGroup.position.set(x, 0, z);

        // Trunk
        const height = 3 + Math.floor(Math.random() * 3);
        for (let y = 0; y < height; y++) {
            treeGroup.add(createVoxel(0, y, 0, COLORS.trunk));
        }

        // Leaves
        const leafColor = type === 'cherry' ? COLORS.cherryPink : COLORS.leafGreen;
        const leafColorAlt = type === 'cherry' ? COLORS.cherryLight : COLORS.leafDark;
        
        const leafSize = type === 'cherry' ? 3 : 2; // Cherry trees are bushier
        
        for (let lx = -leafSize; lx <= leafSize; lx++) {
            for (let lz = -leafSize; lz <= leafSize; lz++) {
                for (let ly = 0; ly < 2; ly++) {
                    // Skip corners to make it rounder
                    if (Math.abs(lx) === leafSize && Math.abs(lz) === leafSize) continue;
                    
                    // Randomly skip some voxels for organic look
                    if (Math.random() > 0.8) continue;

                    const color = Math.random() > 0.5 ? leafColor : leafColorAlt;
                    treeGroup.add(createVoxel(lx, height + ly, lz, color));
                }
            }
        }
        
        // Top cluster
        treeGroup.add(createVoxel(0, height + 2, 0, leafColor));

        scene.add(treeGroup);
    }

    function buildLantern(scene, x, z) {
        const lanternGroup = new THREE.Group();
        lanternGroup.position.set(x, 0, z);

        // Base
        lanternGroup.add(createVoxel(0, 0, 0, COLORS.stoneDark));
        lanternGroup.add(createVoxel(0, 1, 0, COLORS.stoneGray));
        
        // Pillar
        lanternGroup.add(createVoxel(0, 2, 0, COLORS.stoneGray));
        
        // Light Box
        lanternGroup.add(createVoxel(0, 3, 0, COLORS.lanternLight));
        
        // Roof
        lanternGroup.add(createVoxel(0, 4, 0, COLORS.stoneDark));
        lanternGroup.add(createVoxel(1, 4, 0, COLORS.stoneDark));
        lanternGroup.add(createVoxel(-1, 4, 0, COLORS.stoneDark));
        lanternGroup.add(createVoxel(0, 4, 1, COLORS.stoneDark));
        lanternGroup.add(createVoxel(0, 4, -1, COLORS.stoneDark));
        
        // Point light for the lantern
        const light = new THREE.PointLight(0xffeb3b, 0.5, 5);
        light.position.set(0, 3, 0);
        lanternGroup.add(light);

        scene.add(lanternGroup);
    }

    function buildPond(scene) {
        const pondGroup = new THREE.Group();
        
        // Pond Shape (Irregular circle-ish)
        const pondData = [
            [0, 0, 3], [1, 0, 3], [2, 0, 2], [3, 0, 1], [3, 0, 0], [3, 0, -1], [2, 0, -2], [1, 0, -3], [0, 0, -3], [-1, 0, -3], [-2, 0, -2], [-3, 0, -1], [-3, 0, 0], [-3, 0, 1], [-2, 0, 2], [-1, 0, 3]
        ];

        // Water
        pondData.forEach(([x, y, z]) => {
            const water = createVoxel(x, 0, z, COLORS.water);
            water.material.transparent = true;
            water.material.opacity = 0.8;
            pondGroup.add(water);
        });

        // Rocks around the edge
        pondData.forEach(([x, y, z]) => {
            // Add rocks randomly on the border
            if (Math.random() > 0.5) {
                // Check if neighbor is not water to place rock on edge
                const isEdge = !pondData.some(p => p[0] === x+1 && p[2] === z) || 
                               !pondData.some(p => p[0] === x-1 && p[2] === z) ||
                               !pondData.some(p => p[0] === x && p[2] === z+1) ||
                               !pondData.some(p => p[0] === x && p[2] === z-1);
                if (isEdge) {
                    pondGroup.add(createVoxel(x, 0, z, COLORS.stoneGray));
                }
            }
        });

        scene.add(pondGroup);
    }

    function buildGround(scene) {
        const groundGroup = new THREE.Group();
        const size = 20;

        for (let x = -size; x <= size; x++) {
            for (let z = -size; z <= size; z++) {
                // Skip center where pagoda is
                if (Math.abs(x) < 4 && Math.abs(z) < 4) continue;
                
                // Skip pond area roughly
                if (x > -4 && x < 4 && z > -4 && z < 4 && Math.abs(x) + Math.abs(z) < 5) continue;

                let color = COLORS.grass;
                if (Math.random() > 0.8) color = COLORS.grassDark;
                
                // Path to pagoda
                if (Math.abs(x) < 1 || Math.abs(z) < 1) {
                    color = COLORS.path;
                }

                groundGroup.add(createVoxel(x, -1, z, color));
            }
        }
        scene.add(groundGroup);
    }

    // --- POPULATE SCENE ---
    
    buildGround(scene);
    buildPagoda(scene);
    buildPond(scene);

    // Trees
    const treePositions = [
        { x: -8, z: -8, type: 'cherry' },
        { x: 8, z: -8, type: 'cherry' },
        { x: -8, z: 8, type: 'cherry' },
        { x: 8, z: 8, type: 'cherry' },
        { x: -12, z: 0, type: 'green' },
        { x: 12, z: 0, type: 'green' },
        { x: 0, z: -12, type: 'green' },
        { x: 0, z: 12, type: 'green' },
        { x: -6, z: -12, type: 'cherry' },
        { x: 6, z: -12, type: 'green' },
        { x: -12, z: 6, type: 'green' },
        { x: 12, z: 6, type: 'cherry' }
    ];

    treePositions.forEach(t => buildTree(scene, t.x, t.z, t.type));

    // Lanterns along the path
    buildLantern(scene, 3, 3);
    buildLantern(scene, -3, 3);
    buildLantern(scene, 3, -3);
    buildLantern(scene, -3, -3);
    buildLantern(scene, 6, 0);
    buildLantern(scene, -6, 0);

    // --- ANIMATION: FALLING PETALS ---
    const petals = [];
    const petalGeo = new THREE.BoxGeometry(0.3, 0.05, 0.3);
    const petalMat = new THREE.MeshBasicMaterial({ color: COLORS.cherryLight });

    for (let i = 0; i < 100; i++) {
        const petal = new THREE.Mesh(petalGeo, petalMat);
        resetPetal(petal);
        scene.add(petal);
        petals.push(petal);
    }

    function resetPetal(petal) {
        petal.position.x = (Math.random() - 0.5) * 40;
        petal.position.y = 15 + Math.random() * 10;
        petal.position.z = (Math.random() - 0.5) * 40;
        petal.userData = {
            speed: 0.02 + Math.random() * 0.02,
            rotSpeed: (Math.random() - 0.5) * 0.1,
            sway: Math.random() * 2
        };
    }

    // --- ANIMATION LOOP ---
    const clock = new THREE.Clock();

    function animate() {
        requestAnimationFrame(animate);
        const time = clock.getElapsedTime();

        // Update Controls
        controls.update();

        // Animate Petals
        petals.forEach(petal => {
            petal.position.y -= petal.userData.speed;
            petal.position.x += Math.sin(time + petal.userData.sway) * 0.01;
            petal.rotation.x += petal.userData.rotSpeed;
            petal.rotation.z += petal.userData.rotSpeed;

            // Reset if below ground
            if (petal.position.y < -1) {
                resetPetal(petal);
            }
        });

        renderer.render(scene, camera);
    }

    // Handle Resize
    window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    animate();

</script>
</body>
</html>
```
